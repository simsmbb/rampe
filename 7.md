# 🧠 1️⃣ `agentctl` CLI Implementation

## agentctl/cli.py

```python
import click
from deploy_databricks import deploy_databricks
from deploy_ecs import deploy_ecs
from validator import validate_spec
import json

@click.group()
def cli():
    pass

@cli.command()
@click.option("--spec", required=True)
@click.option("--mode", type=click.Choice(["databricks", "ecs"]))
def create(spec, mode):
    with open(spec) as f:
        spec_data = json.load(f)

    validate_spec(spec_data)

    if mode == "databricks":
        deploy_databricks(spec_data)
    else:
        deploy_ecs(spec_data)

if __name__ == "__main__":
    cli()
```

# 🏗 2️⃣ Production SQL Tool (Schema-Aware, Injection Safe)

## runtime/sql_tool.py

```python
import re
from databricks import sql
import os

ALLOWED_SQL = ["SELECT"]

class SQLQueryTool:

    def __init__(self):
        self.host = os.getenv("DB_HOST")
        self.path = os.getenv("DB_WAREHOUSE")
        self.token = os.getenv("DB_TOKEN")

    def validate_sql(self, query):
        query_upper = query.strip().upper()

        if not any(query_upper.startswith(cmd) for cmd in ALLOWED_SQL):
            raise ValueError("Only SELECT statements allowed")

        if re.search(r";|--|DROP|DELETE|INSERT|UPDATE", query_upper):
            raise ValueError("Unsafe SQL detected")

    def run(self, query):
        self.validate_sql(query)

        with sql.connect(
            server_hostname=self.host,
            http_path=self.path,
            access_token=self.token
        ) as conn:
            cursor = conn.cursor()
            cursor.execute(query)
            return cursor.fetchall()
```

# 🤖 3️⃣ Agent Runtime

## runtime/agent_runtime.py

```python
from sql_tool import SQLQueryTool
from model_client import call_llm
from cost_meter import record_usage

class AgentRuntime:

    def __init__(self, spec):
        self.spec = spec
        self.sql_tool = SQLQueryTool()

    def run(self, user_input):

        sql_prompt = f"""
Convert this to SQL for:
main.sales.transactions

{user_input}

Return SQL only.
"""

        sql_query, usage = call_llm(sql_prompt)

        result = self.sql_tool.run(sql_query)

        record_usage(self.spec["agent_name"], usage)

        return result
```

# 💰 4️⃣ Cost Metering Implementation

## runtime/cost_meter.py

```python
import json
from datetime import datetime

def record_usage(agent_name, usage):
    event = {
        "agent": agent_name,
        "input_tokens": usage["input_tokens"],
        "output_tokens": usage["output_tokens"],
        "timestamp": str(datetime.utcnow())
    }

    with open("usage_log.json", "a") as f:
        f.write(json.dumps(event) + "\n")
```

## databricks/mlflow_wrapper.py

```python
import mlflow.pyfunc
from runtime.agent_runtime import AgentRuntime
import json

class AgentModel(mlflow.pyfunc.PythonModel):

    def load_context(self, context):
        with open("spec.json") as f:
            spec = json.load(f)

        self.agent = AgentRuntime(spec)

    def predict(self, context, model_input):
        return self.agent.run(model_input["query"])
```
