# 🧠 Design Principles

1. **Agents are configuration, not code**
2. Must support multi-agent future
3. Must support RBAC & governance
4. Must support tool composition
5. Must be versioned
6. Must support cost & performance policies
7. Must be extensible without breaking older agents

# 🧾 AgentSpec JSON Schema (Full Production Version)

```json
{
  "$schema": "http://json-schema.org/draft-07/schema#",
  "title": "AgentSpec",
  "type": "object",
  "required": [
    "spec_version",
    "agent_id",
    "agent_name",
    "description",
    "owner",
    "model",
    "tools",
    "deployment"
  ],
  "properties": {

    "spec_version": {
      "type": "string",
      "description": "Schema version for backward compatibility",
      "example": "1.0.0"
    },

    "agent_id": {
      "type": "string",
      "description": "Globally unique agent identifier (UUID)"
    },

    "agent_name": {
      "type": "string"
    },

    "description": {
      "type": "string"
    },

    "tags": {
      "type": "array",
      "items": { "type": "string" }
    },

    "owner": {
      "type": "object",
      "required": ["team", "contact_email"],
      "properties": {
        "team": { "type": "string" },
        "contact_email": { "type": "string" },
        "cost_center": { "type": "string" }
      }
    },

    "model": {
      "type": "object",
      "required": ["provider", "name"],
      "properties": {
        "provider": {
          "type": "string",
          "enum": ["openai", "azure_openai", "databricks", "anthropic"]
        },
        "name": { "type": "string" },
        "temperature": { "type": "number", "default": 0.2 },
        "max_tokens": { "type": "integer", "default": 2048 },
        "top_p": { "type": "number", "default": 1.0 }
      }
    },

    "memory": {
      "type": "object",
      "properties": {
        "type": {
          "type": "string",
          "enum": ["none", "conversation_buffer", "vector", "hybrid"]
        },
        "max_history": { "type": "integer", "default": 10 },
        "vector_store": {
          "type": "object",
          "properties": {
            "provider": { "type": "string", "enum": ["databricks_vector"] },
            "index_name": { "type": "string" },
            "embedding_model": { "type": "string" }
          }
        }
      }
    },

    "data_sources": {
      "type": "array",
      "items": {
        "type": "object",
        "required": ["type", "name"],
        "properties": {
          "type": {
            "type": "string",
            "enum": [
              "delta_table",
              "unity_catalog_table",
              "vector_index",
              "external_api",
              "ml_model"
            ]
          },
          "name": { "type": "string" },
          "catalog": { "type": "string" },
          "schema": { "type": "string" },
          "config": { "type": "object" }
        }
      }
    },

    "tools": {
      "type": "array",
      "items": {
        "type": "object",
        "required": ["name", "enabled"],
        "properties": {
          "name": { "type": "string" },
          "enabled": { "type": "boolean" },
          "config": { "type": "object" },
          "timeout_seconds": { "type": "integer", "default": 30 }
        }
      }
    },

    "planner": {
      "type": "object",
      "properties": {
        "strategy": {
          "type": "string",
          "enum": ["single_tool", "multi_step", "react", "chain_of_thought"]
        },
        "max_steps": { "type": "integer", "default": 3 },
        "allow_tool_retry": { "type": "boolean", "default": true }
      }
    },

    "guardrails": {
      "type": "object",
      "properties": {
        "input": {
          "type": "object",
          "properties": {
            "pii_filter": { "type": "boolean" },
            "toxicity_filter": { "type": "boolean" },
            "max_input_tokens": { "type": "integer" }
          }
        },
        "output": {
          "type": "object",
          "properties": {
            "hallucination_check": { "type": "boolean" },
            "citation_required": { "type": "boolean" },
            "toxicity_filter": { "type": "boolean" }
          }
        }
      }
    },

    "response_format": {
      "type": "object",
      "properties": {
        "default_type": {
          "type": "string",
          "enum": ["text", "markdown", "json", "table", "chart"]
        },
        "allow_structured_output": { "type": "boolean" }
      }
    },

    "security": {
      "type": "object",
      "properties": {
        "enforce_unity_catalog_rbac": { "type": "boolean" },
        "row_level_security": { "type": "boolean" },
        "impersonate_request_user": { "type": "boolean" }
      }
    },

    "cost_controls": {
      "type": "object",
      "properties": {
        "max_tokens_per_request": { "type": "integer" },
        "monthly_budget_usd": { "type": "number" },
        "alert_threshold_percent": { "type": "integer" }
      }
    },

    "observability": {
      "type": "object",
      "properties": {
        "log_prompts": { "type": "boolean" },
        "log_tool_calls": { "type": "boolean" },
        "log_responses": { "type": "boolean" },
        "enable_tracing": { "type": "boolean" }
      }
    },

    "deployment": {
      "type": "object",
      "required": ["mode"],
      "properties": {
        "mode": {
          "type": "string",
          "enum": ["databricks_model_serving"]
        },
        "auto_scale": { "type": "boolean" },
        "min_replicas": { "type": "integer" },
        "max_replicas": { "type": "integer" },
        "environment": {
          "type": "string",
          "enum": ["dev", "staging", "prod"]
        }
      }
    }
  }
}
```
