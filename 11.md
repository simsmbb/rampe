# 2️⃣ Billing Data Model (PostgreSQL)

```sql
CREATE TABLE pricing_plans (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL,
    price_per_1k_tokens NUMERIC,
    price_per_query NUMERIC,
    base_subscription_fee NUMERIC
);

CREATE TABLE tenant_subscriptions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id UUID,
    pricing_plan_id UUID REFERENCES pricing_plans(id),
    stripe_subscription_id TEXT,
    active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE usage_events (
    id UUID PRIMARY KEY,
    tenant_id UUID,
    agent_id UUID,
    input_tokens INT,
    output_tokens INT,
    query_count INT,
    execution_time_ms INT,
    created_at TIMESTAMP
);

CREATE TABLE monthly_usage_summary (
    tenant_id UUID,
    month TEXT,
    total_input_tokens BIGINT,
    total_output_tokens BIGINT,
    total_queries BIGINT,
    total_cost NUMERIC,
    PRIMARY KEY (tenant_id, month)
);
```

# 4️⃣ Billing Microservice (FastAPI)

## billing_service/main.py

```python
from fastapi import FastAPI
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
import os
import uuid

DATABASE_URL = os.getenv("DATABASE_URL")

engine = create_engine(DATABASE_URL)
Session = sessionmaker(bind=engine)

app = FastAPI()

@app.post("/usage")
def record_usage(event: dict):

    session = Session()

    session.execute("""
        INSERT INTO usage_events
        (id, tenant_id, agent_id, input_tokens,
         output_tokens, query_count, execution_time_ms, created_at)
        VALUES (:id, :tenant, :agent, :in_t, :out_t,
                :queries, :exec_ms, NOW())
    """, {
        "id": event["event_id"],
        "tenant": event["tenant_id"],
        "agent": event["agent_id"],
        "in_t": event["input_tokens"],
        "out_t": event["output_tokens"],
        "queries": event["query_count"],
        "exec_ms": event["execution_time_ms"]
    })

    session.commit()

    return {"status": "recorded"}
```

# 5️⃣ Cost Calculation Engine

## billing_service/cost_engine.py

```python
def calculate_cost(plan, input_tokens, output_tokens, queries):

    token_cost = (
        (input_tokens + output_tokens) / 1000
    ) * float(plan.price_per_1k_tokens)

    query_cost = queries * float(plan.price_per_query)

    return token_cost + query_cost
```

# 6️⃣ Monthly Aggregation Job

Runs daily (Cron / ECS scheduled task):

```python
from sqlalchemy import text

def aggregate_monthly():

    session = Session()

    session.execute(text("""
        INSERT INTO monthly_usage_summary
        SELECT
            tenant_id,
            to_char(created_at, 'YYYY-MM') as month,
            SUM(input_tokens),
            SUM(output_tokens),
            SUM(query_count),
            0
        FROM usage_events
        GROUP BY tenant_id, month
        ON CONFLICT (tenant_id, month)
        DO UPDATE SET
            total_input_tokens = EXCLUDED.total_input_tokens,
            total_output_tokens = EXCLUDED.total_output_tokens,
            total_queries = EXCLUDED.total_queries
    """))

    session.commit()
```

# 7️⃣ Stripe Integration

Uses Stripe subscription model.

## billing_service/stripe_integration.py

```python
import stripe
import os

stripe.api_key = os.getenv("STRIPE_SECRET_KEY")

def create_customer(email):
    return stripe.Customer.create(email=email)

def create_subscription(customer_id, price_id):
    return stripe.Subscription.create(
        customer=customer_id,
        items=[{"price": price_id}]
    )

def add_usage_record(subscription_item_id, quantity):
    stripe.SubscriptionItem.create_usage_record(
        subscription_item_id,
        quantity=quantity,
        action="increment"
    )
```

# 8️⃣ Stripe Webhook Handler

```python
@app.post("/stripe/webhook")
async def stripe_webhook(request: Request):

    payload = await request.body()
    sig_header = request.headers.get("stripe-signature")

    event = stripe.Webhook.construct_event(
        payload,
        sig_header,
        os.getenv("STRIPE_WEBHOOK_SECRET")
    )

    if event["type"] == "invoice.payment_succeeded":
        print("Payment received")

    return {"status": "ok"}
```
