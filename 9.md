# 1️⃣ ECS Python Program to Access Unity Catalog

Uses Databricks SQL Connector.

## runtime/unity_sql_tool.py

```python
import os
from databricks import sql
import re

ALLOWED_COMMANDS = ["SELECT"]

class UnityCatalogSQLTool:

    def __init__(self):
        self.server_hostname = os.getenv("DB_HOST")
        self.http_path = os.getenv("DB_HTTP_PATH")
        self.access_token = os.getenv("DB_TOKEN")

    def _validate(self, query: str):
        q = query.strip().upper()

        if not any(q.startswith(cmd) for cmd in ALLOWED_COMMANDS):
            raise ValueError("Only SELECT queries allowed")

        if re.search(r";|DROP|DELETE|UPDATE|INSERT", q):
            raise ValueError("Unsafe SQL detected")

    def execute(self, query: str):
        self._validate(query)

        with sql.connect(
            server_hostname=self.server_hostname,
            http_path=self.http_path,
            access_token=self.access_token
        ) as connection:

            cursor = connection.cursor()
            cursor.execute(query)
            return cursor.fetchall()
```

# 4️⃣ Agent Runtime

## runtime/llm_client.py

```python
from openai import OpenAI
import os

client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))

def generate_sql(prompt, model="gpt-4o-mini"):
    response = client.chat.completions.create(
        model=model,
        messages=[
            {"role": "system", "content": "You generate safe SQL only."},
            {"role": "user", "content": prompt}
        ]
    )

    sql_query = response.choices[0].message.content
    usage = response.usage

    return sql_query, usage
```

## runtime/agent_runtime.py

```python
import json
from unity_sql_tool import UnityCatalogSQLTool
from llm_client import generate_sql

class AgentRuntime:

    def __init__(self, spec_path: str):
        with open(spec_path) as f:
            self.spec = json.load(f)

        self.sql_tool = UnityCatalogSQLTool()

    def build_prompt(self, question: str):

        tables = ", ".join(
            self.spec["data_access"]["allowed_tables"]
        )

        return f"""
You are an analytics assistant.

Allowed tables: {tables}

Question:
{question}

Return SQL only.
"""

    def run(self, question: str):

        prompt = self.build_prompt(question)

        sql_query, usage = generate_sql(prompt)

        result = self.sql_tool.execute(sql_query)

        return {
            "sql": sql_query,
            "rows": result,
            "usage": {
                "input_tokens": usage.prompt_tokens,
                "output_tokens": usage.completion_tokens
            }
        }
```

## publisher/deploy_databricks.py

```python
import requests
import os

def create_serving_endpoint(model_name):

    url = f"{os.getenv('DATABRICKS_HOST')}/api/2.0/serving-endpoints"

    payload = {
        "name": model_name,
        "config": {
            "served_models": [{
                "model_name": model_name,
                "model_version": "1",
                "workload_size": "Small"
            }]
        }
    }

    headers = {
        "Authorization": f"Bearer {os.getenv('DATABRICKS_TOKEN')}"
    }

    requests.post(url, json=payload, headers=headers)
```
