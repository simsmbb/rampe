## 📦 PostgreSQL Schema

```sql
CREATE TABLE tenants (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL,
    tier TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE agents (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id UUID REFERENCES tenants(id),
    name TEXT NOT NULL,
    description TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE agent_versions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    agent_id UUID REFERENCES agents(id),
    version_number INTEGER NOT NULL,
    spec JSONB NOT NULL,
    status TEXT DEFAULT 'draft',
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE deployments (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    agent_version_id UUID REFERENCES agent_versions(id),
    environment TEXT NOT NULL,
    platform TEXT NOT NULL, -- ecs | databricks
    endpoint_url TEXT,
    status TEXT,
    deployed_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE usage_records (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    agent_id UUID REFERENCES agents(id),
    tenant_id UUID REFERENCES tenants(id),
    input_tokens INT,
    output_tokens INT,
    execution_time_ms INT,
    created_at TIMESTAMP DEFAULT NOW()
);
```

# 🧩 Registry Service (FastAPI)

## registry_service/main.py

```python
from fastapi import FastAPI
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
import os
import json
import uuid

DATABASE_URL = os.getenv("DATABASE_URL")

engine = create_engine(DATABASE_URL)
Session = sessionmaker(bind=engine)

app = FastAPI()

@app.post("/agents")
def create_agent(payload: dict):
    session = Session()

    agent_id = str(uuid.uuid4())
    version_id = str(uuid.uuid4())

    session.execute("""
        INSERT INTO agents (id, tenant_id, name, description)
        VALUES (:id, :tenant, :name, :desc)
    """, {
        "id": agent_id,
        "tenant": payload["tenant_id"],
        "name": payload["name"],
        "desc": payload["description"]
    })

    session.execute("""
        INSERT INTO agent_versions (id, agent_id, version_number, spec, status)
        VALUES (:vid, :aid, 1, :spec, 'draft')
    """, {
        "vid": version_id,
        "aid": agent_id,
        "spec": json.dumps(payload["spec"])
    })

    session.commit()

    return {"agent_id": agent_id, "version_id": version_id}
```

# 🔁 Agent Version Promotion Logic

```python
@app.post("/agents/{version_id}/publish")
def publish_agent(version_id: str):
    session = Session()

    session.execute("""
        UPDATE agent_versions
        SET status='published'
        WHERE id=:vid
    """, {"vid": version_id})

    session.commit()

    return {"status": "published"}
```

# 📁 .github/workflows/ci-cd.yml

```yaml
name: Agent Platform CI/CD

on:
  push:
    branches: [ main ]
  pull_request:

env:
  AWS_REGION: us-east-1
  ECR_REPOSITORY: agent-runtime
  ECS_SERVICE: agent-service
  ECS_CLUSTER: agent-cluster

jobs:

  test:
    runs-on: ubuntu-latest

    steps:
      - uses: actions/checkout@v3

      - name: Setup Python
        uses: actions/setup-python@v4
        with:
          python-version: 3.11

      - name: Install dependencies
        run: |
          pip install -r ecs_service/requirements.txt
          pip install pytest flake8 bandit

      - name: Lint
        run: flake8 .

      - name: Security Scan
        run: bandit -r .

      - name: Run Tests
        run: pytest tests/

  build-and-deploy:
    needs: test
    runs-on: ubuntu-latest

    permissions:
      id-token: write
      contents: read

    steps:

      - uses: actions/checkout@v3

      - name: Configure AWS Credentials
        uses: aws-actions/configure-aws-credentials@v3
        with:
          role-to-assume: ${{ secrets.AWS_ROLE }}
          aws-region: ${{ env.AWS_REGION }}

      - name: Login to ECR
        uses: aws-actions/amazon-ecr-login@v2

      - name: Build Docker Image
        run: |
          docker build -t $ECR_REPOSITORY:latest .
          docker tag $ECR_REPOSITORY:latest \
            ${{ secrets.AWS_ACCOUNT_ID }}.dkr.ecr.$AWS_REGION.amazonaws.com/$ECR_REPOSITORY:latest

      - name: Push to ECR
        run: |
          docker push \
          ${{ secrets.AWS_ACCOUNT_ID }}.dkr.ecr.$AWS_REGION.amazonaws.com/$ECR_REPOSITORY:latest

      - name: Deploy to ECS
        run: |
          aws ecs update-service \
            --cluster $ECS_CLUSTER \
            --service $ECS_SERVICE \
            --force-new-deployment

      - name: Deploy to Databricks
        if: github.ref == 'refs/heads/main'
        run: |
          curl -X POST \
          -H "Authorization: Bearer ${{ secrets.DATABRICKS_TOKEN }}" \
          -H "Content-Type: application/json" \
          https://${{ secrets.DATABRICKS_HOST }}/api/2.0/serving-endpoints
```
