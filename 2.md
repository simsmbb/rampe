# 3️⃣ Brick Loader

Loads reusable components dynamically.

`brick_loader.py`

```python
from tool_registry.registry import TOOL_REGISTRY


class BrickLoader:
    def __init__(self, spec):
        self.spec = spec

    def load_tools(self):
        tools = []
        for tool_name in self.spec.tools:
            if tool_name not in TOOL_REGISTRY:
                raise ValueError(f"Tool {tool_name} not registered")
            tools.append(TOOL_REGISTRY[tool_name]())
        return tools
```

# 5️⃣ Tool Executor

`tool_executor.py`

```python
class ToolExecutor:
    def __init__(self, tools):
        self.tools = {tool.name: tool for tool in tools}

    def execute(self, plan, user_input):
        tool_name = plan["tool"]

        if tool_name not in self.tools:
            raise ValueError("Invalid tool selected")

        return self.tools[tool_name].run(user_input)
```

# 9️⃣ Core Agent Runtime

`runtime.py`

```python
from planner import Planner
from tool_executor import ToolExecutor
from memory import MemoryManager
from guardrails import GuardrailEngine
from brick_loader import BrickLoader
from formatter import ResponseFormatter


class AgentRuntime:

    def __init__(self, spec, model_client):
        self.spec = spec
        self.model_client = model_client

        self.brick_loader = BrickLoader(spec)
        self.tools = self.brick_loader.load_tools()

        self.planner = Planner(model_client)
        self.executor = ToolExecutor(self.tools)
        self.memory = MemoryManager(spec.memory)
        self.guardrails = GuardrailEngine(spec.guardrails)
        self.formatter = ResponseFormatter()

    def run(self, user_input, context):
        # Input guardrails
        self.guardrails.validate_input(user_input)

        # Plan
        plan = self.planner.create_plan(user_input, self.tools)

        # Execute
        result = self.executor.execute(plan, user_input)

        # Output guardrails
        self.guardrails.validate_output(str(result))

        # Memory update
        self.memory.add(user_input, result)

        # Format response
        return self.formatter.format(result)
```

        model_client = load_model_client(spec.model)

        self.agent = AgentRuntime(spec, model_client)

    def predict(self, context, model_input):
