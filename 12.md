## Historical (Postgres)

```sql
CREATE TABLE usage_hourly (
    tenant_id UUID,
    hour TIMESTAMP,
    input_tokens BIGINT,
    output_tokens BIGINT,
    queries BIGINT,
    cost NUMERIC,
    PRIMARY KEY (tenant_id, hour)
);
```

# 3️⃣ Real-Time Ingestion Update

Modify billing service `/usage`:

```python
import redis
r = redis.Redis(host="redis", port=6379)

@app.post("/usage")
def record_usage(event: dict):

    # Store in Postgres (existing logic)

    total_tokens = event["input_tokens"] + event["output_tokens"]

    r.incrby(f"tenant:{event['tenant_id']}:tokens_today", total_tokens)
    r.incr(f"tenant:{event['tenant_id']}:queries_today", 1)

    r.publish(
        f"tenant:{event['tenant_id']}:live_stream",
        str(total_tokens)
    )

    return {"status": "ok"}
```

# 4️⃣ Dashboard API Service

## dashboard_service/main.py

```python
from fastapi import FastAPI, WebSocket
import redis
import os

app = FastAPI()

r = redis.Redis(host=os.getenv("REDIS_HOST"), port=6379)

@app.get("/usage/today/{tenant_id}")
def get_today_usage(tenant_id: str):

    tokens = r.get(f"tenant:{tenant_id}:tokens_today") or 0
    queries = r.get(f"tenant:{tenant_id}:queries_today") or 0

    return {
        "tokens_today": int(tokens),
        "queries_today": int(queries)
    }

@app.websocket("/ws/{tenant_id}")
async def websocket_endpoint(websocket: WebSocket, tenant_id: str):

    await websocket.accept()

    pubsub = r.pubsub()
    pubsub.subscribe(f"tenant:{tenant_id}:live_stream")

    for message in pubsub.listen():
        if message["type"] == "message":
            await websocket.send_text(message["data"].decode())
```

# 5️⃣ React Frontend

## frontend/src/App.js

```javascript
import React, { useEffect, useState } from "react";
import { LineChart, Line, XAxis, YAxis, Tooltip } from "recharts";

function App() {

  const tenantId = "TENANT_ID";
  const [tokens, setTokens] = useState(0);
  const [queries, setQueries] = useState(0);
  const [liveData, setLiveData] = useState([]);

  useEffect(() => {
    fetch(`/usage/today/${tenantId}`)
      .then(res => res.json())
      .then(data => {
        setTokens(data.tokens_today);
        setQueries(data.queries_today);
      });

    const ws = new WebSocket(`ws://localhost:8000/ws/${tenantId}`);

    ws.onmessage = (event) => {
      setLiveData(prev => [
        ...prev,
        { time: new Date().toLocaleTimeString(), tokens: Number(event.data) }
      ]);
    };

  }, []);

  return (
    <div>
      <h1>Usage Dashboard</h1>
      <p>Tokens Today: {tokens}</p>
      <p>Queries Today: {queries}</p>

      <LineChart width={600} height={300} data={liveData}>
        <XAxis dataKey="time"/>
        <YAxis/>
        <Tooltip/>
        <Line type="monotone" dataKey="tokens"/>
      </LineChart>
    </div>
  );
}

export default App;
```
